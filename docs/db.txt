-- ============================================================================
-- SECTION 1: USER & AUTHENTICATION MODULE
-- ============================================================================

CREATE TABLE `user` (
    `user_id`               INT             NOT NULL AUTO_INCREMENT,
    `username`              VARCHAR(50)     NOT NULL,
    `password_hash`         VARCHAR(255)    NOT NULL    COMMENT 'BCrypt encrypted password',
    `full_name`             VARCHAR(100)    NOT NULL,
    `email`                 VARCHAR(150)    NULL,
    `phone`                 VARCHAR(20)     NULL,
    `role`                  ENUM('ADMIN', 'TRAINER', 'CONTENT_EDITOR', 'REVIEWER')
                                            NOT NULL,
    `military_rank`         VARCHAR(50)     NULL        COMMENT 'Cấp bậc quân sự (Trainer)',
    `unit`                  VARCHAR(100)    NULL        COMMENT 'Đơn vị phụ trách (Trainer)',
    `is_active`             TINYINT(1)      NOT NULL DEFAULT 1,
    `is_locked`             TINYINT(1)      NOT NULL DEFAULT 0  COMMENT 'Khóa sau 5 lần đăng nhập sai',
    `failed_login_count`    INT             NOT NULL DEFAULT 0,
    `last_login_at`         DATETIME        NULL,
    `created_at`            DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    `updated_at`            DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    `is_deleted`            TINYINT(1)      NOT NULL DEFAULT 0,
    `deleted_at`            DATETIME        NULL,

    PRIMARY KEY (`user_id`),
    UNIQUE KEY `uk_user_username` (`username`),
    INDEX `idx_user_role` (`role`),
    INDEX `idx_user_active` (`is_active`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
  COMMENT='Bảng user trung tâm cho tất cả actors: Admin, Trainer, Editor, Reviewer';

-- Gộp breed_weight_standard vào đây luôn (weight_min/max theo giống)
-- Dùng metadata JSON cho các trường mở rộng
CREATE TABLE `dog_breed` (
    `breed_id`              INT             NOT NULL AUTO_INCREMENT,
    `breed_name`            VARCHAR(100)    NOT NULL,
    `origin`                VARCHAR(100)    NULL,
    `description`           TEXT            NULL,
    `size_classification`   ENUM('SMALL', 'MEDIUM', 'LARGE', 'GIANT')
                                            NULL,
    -- Weight standard gộp trực tiếp (thay breed_weight_standard)
    `weight_male_min_kg`    DECIMAL(5,2)    NULL        COMMENT 'Cân nặng chuẩn tối thiểu (đực)',
    `weight_male_max_kg`    DECIMAL(5,2)    NULL        COMMENT 'Cân nặng chuẩn tối đa (đực)',
    `weight_female_min_kg`  DECIMAL(5,2)    NULL        COMMENT 'Cân nặng chuẩn tối thiểu (cái)',
    `weight_female_max_kg`  DECIMAL(5,2)    NULL        COMMENT 'Cân nặng chuẩn tối đa (cái)',
    `avg_height_cm`         DECIMAL(5,2)    NULL,
    `lifespan_years`        VARCHAR(20)     NULL        COMMENT 'VD: 10-13',
    `trainability_level`    ENUM('LOW', 'MEDIUM', 'HIGH', 'VERY_HIGH')
                                            NULL,
    `operational_capabilities` TEXT         NULL        COMMENT 'Khả năng tác chiến quân sự',
    -- Metadata JSON cho các trường có thể mở rộng sau
    `metadata`              JSON            NULL        COMMENT 'Mở rộng: {behavioral_traits, temperament, care_requirements, physical_characteristics}',
    `image_url`             VARCHAR(500)    NULL,
    `status`                ENUM('DRAFT', 'PENDING', 'APPROVED', 'PUBLISHED', 'REJECTED')
                                            NOT NULL DEFAULT 'DRAFT',
    `created_by`            INT             NULL,
    `created_at`            DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    `updated_at`            DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    `is_deleted`            TINYINT(1)      NOT NULL DEFAULT 0,
    `deleted_at`            DATETIME        NULL,

    PRIMARY KEY (`breed_id`),
    UNIQUE KEY `uk_breed_name` (`breed_name`),
    INDEX `idx_breed_size` (`size_classification`),
    INDEX `idx_breed_status` (`status`),
    CONSTRAINT `fk_breed_created_by` FOREIGN KEY (`created_by`) REFERENCES `user`(`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
  COMMENT='Giống chó quân sự với thông số chuẩn cân nặng theo giới tính';
